const SNIPPET_KEY='arcodes-snippets-v1';
function loadSnippets(){try{const v=JSON.parse(localStorage.getItem(SNIPPET_KEY)||'[]');return Array.isArray(v)?v:[]}catch{return []}}
function saveSnippets(items){try{localStorage.setItem(SNIPPET_KEY,JSON.stringify(items));return true}catch{return false}}
function renderSnippets(){const root=$('snippet-list');root.replaceChildren();const items=loadSnippets();$('snippet-count').textContent=items.length+' saved snippet'+(items.length===1?'':'s')+' in this browser';if(!items.length){const empty=document.createElement('div');empty.className='empty';empty.textContent='No snippets saved yet. Add your first reusable snippet above.';root.append(empty);return}items.forEach(item=>{const card=document.createElement('article');card.className='tool-card';card.style.minHeight='0';const title=document.createElement('h3');title.textContent=item.title;const meta=document.createElement('p');meta.textContent=item.language+' · '+new Date(item.createdAt).toLocaleDateString();const preview=document.createElement('pre');preview.className='result';preview.style.minHeight='0';preview.style.maxHeight='140px';preview.textContent=item.code;const actions=document.createElement('div');actions.className='work-actions';const copy=document.createElement('button');copy.className='small-btn';copy.textContent='Copy code';copy.onclick=()=>{if(navigator.clipboard&&navigator.clipboard.writeText)navigator.clipboard.writeText(item.code).then(()=>toast('Snippet copied')).catch(()=>toast('Clipboard permission unavailable'));else toast('Clipboard unavailable in this browser')};const remove=document.createElement('button');remove.className='small-btn';remove.textContent='Delete';remove.onclick=()=>{const next=loadSnippets().filter(x=>x.id!==item.id);if(saveSnippets(next)){renderSnippets();toast('Snippet deleted')}else toast('Browser storage is unavailable')};actions.append(copy,remove);card.append(title,meta,preview,actions);root.append(card)})}
