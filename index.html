const SNIPPET_KEY='arcodes-snippets-v1';
function loadSnippets(){try{const v=JSON.parse(localStorage.getItem(SNIPPET_KEY)||'[]');return Array.isArray(v)?v:[]}catch{return []}}
function saveSnippets(items){try{localStorage.setItem(SNIPPET_KEY,JSON.stringify(items));return true}catch{return false}}
